"""SQLite storage for taskcli. Tasks live in tasks.db (table `tasks`) in
the current working directory. tasks.json is no longer written or read.
"""
import sqlite3

DB_FILE = "tasks.db"


def get_connection(path=DB_FILE):
    conn = sqlite3.connect(path)
    conn.execute("""
        CREATE TABLE IF NOT EXISTS tasks (
            id INTEGER PRIMARY KEY,
            title TEXT NOT NULL,
            done INTEGER NOT NULL DEFAULT 0,
            priority TEXT NOT NULL DEFAULT 'med',
            due TEXT
        )
    """)
    conn.commit()
    return conn


def load_tasks(path=DB_FILE):
    conn = get_connection(path)
    try:
        rows = conn.execute(
            "SELECT id, title, done, priority, due FROM tasks ORDER BY id"
        ).fetchall()
    finally:
        conn.close()
    tasks = []
    for id_, title, done, priority, due in rows:
        task = {"id": id_, "title": title, "done": bool(done), "priority": priority}
        if due:
            task["due"] = due
        tasks.append(task)
    return tasks


def save_tasks(tasks, path=DB_FILE):
    conn = get_connection(path)
    try:
        conn.execute("DELETE FROM tasks")
        for t in tasks:
            conn.execute(
                "INSERT INTO tasks (id, title, done, priority, due) VALUES (?, ?, ?, ?, ?)",
                (
                    t["id"],
                    t["title"],
                    int(t.get("done", False)),
                    t.get("priority", "med"),
                    t.get("due"),
                ),
            )
        conn.commit()
    finally:
        conn.close()


def next_id(tasks):
    if not tasks:
        return 1
    return max(t["id"] for t in tasks) + 1
